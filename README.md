        function selectQuoteCategory(catName) {
            const select = document.getElementById('qCategory');
            if (!select) return;
            for (let i = 0; i < select.options.length; i++) {
                if (select.options[i].value === catName) {

                    break;
                }
            }
        }

        /**
         * Điều khiển 2 Banner tự trượt & hỗ trợ vuốt trên điện thoại
         */
        let currentSlideIndex = 0;
        let slideTimer = null;

        function updateSliderUI() {
            const slides = document.querySelectorAll('#heroSlider .hero-slide');
            const dots = document.querySelectorAll('.slider-dot');
            slides.forEach((slide, idx) => {
                slide.classList.toggle('active', idx === currentSlideIndex);
            });
            dots.forEach((dot, idx) => {
                if (idx === currentSlideIndex) {
                    dot.className = 'slider-dot w-7 h-2 rounded-full bg-brand-primary transition-all';
                } else {
                    dot.className = 'slider-dot w-2.5 h-2 rounded-full bg-slate-300 transition-all';
                }
            });
        }

        function changeSlide(step) {
            const slides = document.querySelectorAll('#heroSlider .hero-slide');
            if (!slides.length) return;
            currentSlideIndex = (currentSlideIndex + step + slides.length) % slides.length;
            updateSliderUI();
            resetAutoSlide();
        }

        function goToSlide(index) {
            currentSlideIndex = index;
            updateSliderUI();
            resetAutoSlide();
        }

        function resetAutoSlide() {
            if (slideTimer) clearInterval(slideTimer);
            slideTimer = setInterval(() => {
                changeSlide(1);
            }, 5500);
        }

        /**
         * Menu điện thoại & Accordion mục con
         */
        function toggleSubmenu(subId, btn) {
            const el = document.getElementById(subId);
            if (!el) return;
            el.classList.toggle('hidden');
            const svg = btn.querySelector('svg');
            if (svg) svg.classList.toggle('rotate-180');
        }

        /**
         * Khởi tạo toàn bộ sự kiện khi trang tải xong
         */
        document.addEventListener('DOMContentLoaded', function () {
            // Render sản phẩm và bài viết
            renderProducts('all');
            renderArticles('all');

            // Chạy banner tự trượt
            updateSliderUI();
            resetAutoSlide();

            // Hỗ trợ vuốt cảm ứng (swipe) cho Banner trên điện thoại
            const heroSlider = document.getElementById('heroSlider');
            let touchStartX = 0;
            if (heroSlider) {
                heroSlider.addEventListener('touchstart', (e) => {
                    touchStartX = e.changedTouches[0].screenX;
                }, { passive: true });

                heroSlider.addEventListener('touchend', (e) => {
                    const touchEndX = e.changedTouches[0].screenX;
                    if (touchStartX - touchEndX > 45) changeSlide(1);
                    if (touchEndX - touchStartX > 45) changeSlide(-1);
                }, { passive: true });
            }

            // Bật/tắt menu di động
            const toggleBtn = document.getElementById('mobileMenuToggle');
            const panel = document.getElementById('mobileMenuPanel');
            const iconOpen = document.getElementById('iconMenuOpen');
            const iconClose = document.getElementById('iconMenuClose');

            function closeMobileMenu() {
                if (!panel) return;
                panel.classList.add('hidden');
                iconOpen.classList.remove('hidden');
                iconClose.classList.add('hidden');
                toggleBtn.setAttribute('aria-expanded', 'false');
            }

            if (toggleBtn && panel) {
                toggleBtn.addEventListener('click', () => {
                    const isHidden = panel.classList.contains('hidden');
                    panel.classList.toggle('hidden');
                    iconOpen.classList.toggle('hidden', isHidden);
                    iconClose.classList.toggle('hidden', !isHidden);
                    toggleBtn.setAttribute('aria-expanded', String(isHidden));
                });

                document.querySelectorAll('.mobile-nav-close').forEach(link => {
                    link.addEventListener('click', closeMobileMenu);
                });
            }

            // Đóng modal khi bấm ra ngoài nền tối
            const detailModal = document.getElementById('detailModal');
            if (detailModal) {
                detailModal.addEventListener('click', (e) => {
                    if (e.target === detailModal) closeModal();
                });
            }

            // Xử lý gửi Form yêu cầu báo giá (dùng hộp thông báo nội tuyến, không dùng alert)
            const quoteForm = document.getElementById('quoteForm');
            const quoteFeedback = document.getElementById('quoteFeedback');
            if (quoteForm && quoteFeedback) {
                quoteForm.addEventListener('submit', function (e) {
                    e.preventDefault();
                    const name = document.getElementById('qName').value.trim();
                    const phone = document.getElementById('qPhone').value.trim();
                    const category = document.getElementById('qCategory').value;

                    if (!name || !phone) {
                        quoteFeedback.className = 'rounded-lg p-3 text-xs sm:text-sm font-medium bg-red-50 text-red-700 border border-red-200';
                        quoteFeedback.textContent = 'Vui lòng nhập đầy đủ Họ tên và Số điện thoại để Nhôm Kính Đông Huy liên hệ báo giá.';
                        quoteFeedback.classList.remove('hidden');
                        return;
                    }

                    quoteFeedback.className = 'rounded-lg p-3 text-xs sm:text-sm font-medium bg-emerald-50 text-emerald-800 border border-emerald-200';
                    quoteFeedback.innerHTML = 'Cảm ơn <strong>' + name + '</strong>! Yêu cầu báo giá <strong>' + category + '</strong> đã được ghi nhận. Kỹ thuật viên sẽ liên hệ số <strong>' + phone + '</strong> hoặc bạn có thể gọi trực tiếp <strong>0915.837.343</strong>.';
                    quoteFeedback.classList.remove('hidden');
                    quoteForm.reset();
                });
            }
